ALGORITHM 1: HCS MERKLE TREE BATCHING & IPFS CID CHUNKING
================================================================================
Purpose:
  Enables bulk upload of N historical Electronic Health Records (EHRs) by 
  compressing N document CIDs/hashes into a single cryptographic Merkle Root. 
  The Merkle Root is anchored onto Hedera Consensus Service (HCS Topic 0.0.4891024) 
  in 1 single transaction instead of N transactions, cutting Hedera gas/network 
  fees by 92.4%+ and preventing ledger congestion.

--------------------------------------------------------------------------------
PSEUDOCODE:
--------------------------------------------------------------------------------

ALGORITHM ComputeMerkleBatch(CIDs)
INPUT:
    CIDs : Array of string IPFS Content Identifiers [CID_1, CID_2, ..., CID_N]
OUTPUT:
    MerkleRoot : Hexadecimal string representation of top root (0x...)
    LeafHashes : Array of SHA-256 leaf hashes
    TreeDepth  : Integer height of Merkle Tree
    LeafCount  : Total number of leaf nodes N

PRECONDITIONS:
    - CIDs is an array of valid IPFS multihashes or document hashes.

BEGIN
    // Step 1: Handle base/empty edge case
    IF CIDs is NULL OR Length(CIDs) == 0 THEN
        EmptyRoot := SHA256("EMPTY_MERKLE_TREE")
        RETURN {
            MerkleRoot: "0x" + EmptyRoot,
            LeafHashes: [],
            TreeDepth: 0,
            LeafCount: 0
        }
    END IF

    // Step 2: Compute cryptographic SHA-256 hashes for all leaf nodes
    CurrentLevel := []
    FOR EACH cid IN CIDs DO
        LeafHash := SHA256(cid)
        APPEND LeafHash TO CurrentLevel
    END FOR

    LeafHashes := COPY(CurrentLevel)
    TreeDepth := 1

    // Step 3: Iteratively build the Merkle Tree upwards to a single root
    WHILE Length(CurrentLevel) > 1 DO
        NextLevel := []
        Index := 0

        WHILE Index < Length(CurrentLevel) DO
            // Case A: Pair of sibling nodes exists
            IF Index + 1 < Length(CurrentLevel) THEN
                LeftChild := CurrentLevel[Index]
                RightChild := CurrentLevel[Index + 1]
                ParentHash := SHA256(LeftChild + RightChild)
                APPEND ParentHash TO NextLevel
                Index := Index + 2
            // Case B: Odd node at end of level (duplicate node per standard Merkle tree spec)
            ELSE
                SoloChild := CurrentLevel[Index]
                ParentHash := SHA256(SoloChild + SoloChild)
                APPEND ParentHash TO NextLevel
                Index := Index + 1
            END IF
        END WHILE

        CurrentLevel := NextLevel
        TreeDepth := TreeDepth + 1
    END WHILE

    // Step 4: Extract the top-level Merkle Root commitment
    MerkleRoot := "0x" + CurrentLevel[0]

    RETURN {
        MerkleRoot: MerkleRoot,
        LeafHashes: LeafHashes,
        TreeDepth: TreeDepth,
        LeafCount: Length(CIDs)
    }
END ALGORITHM

INPUT:
    PatientAddress : 42-character hex wallet address of patient

OUTPUT:
    TransactionResult : HCS Submission receipt containing Consensus Timestamp and Sequence Number

BEGIN
    // Step 1: Generate off-chain Merkle Root

    RETURN Receipt
END ALGORITHM

STRUCTURE Event:
    Payload             : Array of User Transactions / HCS Topic Messages
    SelfParentHash      : Cryptographic hash of creator's previous event
    OtherParentHash     : Cryptographic hash of receiver's latest event
    CreatorID           : Node public key / node ID
    CreatorSignature    : Digital signature of creator node
    CreationTimestamp   : Local clock timestamp of creator node
    ConsensusTimestamp  : Median consensus timestamp (assigned after virtual voting)
    ConsensusOrder      : Assigned global sequence number
END STRUCTURE

ALGORITHM GossipSyncRound(NodeA, NodeB)
INPUT:
    NodeA : Calling node
    NodeB : Randomly selected peer node in Hedera network
OUTPUT:
    NewEventsSynchronized : Boolean indicating sync completion

BEGIN
    // Step 1: Query events known by NodeB that NodeA does not have
    MissingEvents := NodeB.GetEventsNotKnownBy(NodeA.GetSyncState())

    // Step 2: Transmit missing events over TLS to NodeA
    NodeA.ReceiveAndValidateEvents(MissingEvents)

    // Step 3: NodeA creates a new event packaging local transactions and parents
    NewEvent := CREATE Event(
        Payload: NodeA.GetPendingTransactions(),
        SelfParentHash: NodeA.GetLatestSelfEventHash(),
        OtherParentHash: NodeB.GetLatestEventHash(),
        CreatorID: NodeA.NodeID,
        CreationTimestamp: CurrentLocalTimestamp()
    )
    NewEvent.CreatorSignature := Sign(NewEvent, NodeA.PrivateKey)
    NodeA.AddToLocalDAG(NewEvent)

    // Step 4: Run Virtual Voting on DAG to calculate consensus order
    RunVirtualVotingConsensus(NodeA.LocalDAG)

    RETURN TRUE
END ALGORITHM


ALGORITHM RunVirtualVotingConsensus(DAG)
INPUT:
    DAG : Directed Acyclic Graph of gossip events
OUTPUT:
    OrderedEvents : Linearly ordered, immutable ledger of consensus events

BEGIN
    // Step 1: Divide DAG events into Rounds
    FOR EACH Event e IN DAG WHERE e.Round IS NOT DEFINED DO
        r := MAX(e.SelfParent.Round, e.OtherParent.Round)
        
        // Check if event e can "strongly see" a 2/3 supermajority of witnesses in round r
        WitnessCount := CountStronglySeenWitnesses(DAG, e, r)
        IF WitnessCount > (2/3 * TotalNetworkNodes) THEN
            e.Round := r + 1
        ELSE
            e.Round := r
        END IF

        // Mark whether e is the first event created by its author in this round
        IF IsFirstEventInRoundByAuthor(DAG, e, e.Round) THEN
            e.IsWitness := TRUE
        ELSE
            e.IsWitness := FALSE
        END IF
    END FOR

    // Step 2: Determine Famous Witnesses using Virtual Voting (Zero Message Passing)
    FOR EACH Witness w IN DAG WHERE w.IsWitness == TRUE AND w.IsFamous IS NOT DECIDED DO
        FOR EACH LaterRound r FROM (w.Round + 1) TO DAG.MaxRound DO
            VotesForFamous := 0
            VotesAgainstFamous := 0

            FOR EACH Witness y IN Round r DO
                Vote := DetermineVirtualVote(DAG, y, w)
                IF Vote == TRUE THEN
                    VotesForFamous := VotesForFamous + 1
                ELSE
                    VotesAgainstFamous := VotesAgainstFamous + 1
                END IF
            END FOR

            // If supermajority (2/3) reached, decide fame permanently
            IF VotesForFamous > (2/3 * TotalNetworkNodes) THEN
                w.IsFamous := TRUE
                BREAK
            ELSE IF VotesAgainstFamous > (2/3 * TotalNetworkNodes) THEN
                w.IsFamous := FALSE
                BREAK
            END IF
        END FOR
    END FOR

    // Step 3: Calculate Consensus Timestamp & Sequence Order for Events
    FOR EACH Event e IN DAG WHERE e.ConsensusOrder IS UNDEFINED DO
        // Event e is received in Round R if all famous witnesses in Round R see e
        IF AllFamousWitnessesSeeEvent(DAG, e, TargetRound) THEN
            // Consensus timestamp is median of timestamps when famous witnesses first saw e
            SeeingTimestamps := CollectFamousWitnessReceiveTimestamps(DAG, e, TargetRound)
            e.ConsensusTimestamp := MEDIAN(SeeingTimestamps)
            
            // Assign deterministic global consensus order
            APPEND e TO OrderedConsensusQueue
        END IF
    END FOR

    SORT OrderedConsensusQueue BY ConsensusTimestamp ASC, WhiteNoiseTieBreaker ASC
    RETURN OrderedConsensusQueue
END ALGORITHM

1. MERKLE TREE HASHING & PROOF REPRESENTATIONS
================================================================================

1.1 Leaf Hash Formula:
    For a set of N IPFS Content Identifiers {CID_1, CID_2, ..., CID_N}:
    
        Leaf_i = SHA256( CID_i )        for all i in {1, 2, ..., N}

1.2 Parent Node Combination:
    For adjacent sibling nodes at depth d:
    
        Parent(Node_left, Node_right) = SHA256( Node_left || Node_right )
        
    Where || denotes byte-level string concatenation.

1.3 Odd-Node Sibling Balancing (Merkle Tree Spec):
    When the number of nodes at level k is odd:
    
        Parent(Node_solo) = SHA256( Node_solo || Node_solo )

1.4 Merkle Root Anchor:
    
        MerkleRoot = SHA256( CID_1 || CID_2 || ... || CID_N )

1.5 Binary Tree Height and Depth:
    
        TreeHeight = ceil( log2( N ) )

1.6 Merkle Audit Proof Size:
    
        ProofSize = O( log2( N ) )

1.7 Transaction Cost Optimization Equation:
    Let C_tx be the base cost of 1 Hedera HCS message ($0.0001 USD).
    Cost for unbatched individual submissions:
        Cost_individual = N * C_tx
    Cost for Merkle batched submission:
        Cost_batched = 1 * C_tx
    Percentage Savings:
        Savings = [ ( Cost_individual - Cost_batched ) / Cost_individual ] * 100%
        Savings = [ ( N - 1 ) / N ] * 100%
    For N = 13:
        Savings = (12 / 13) * 100% = 92.31% (~92.4%)
